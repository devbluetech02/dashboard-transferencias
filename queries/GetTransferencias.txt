SELECT
  t.TRANSACAO,

  CASE
    WHEN t.CODFILIALORIG IN (1, 11, 5, 51)    THEN 'GO'
    WHEN t.CODFILIALORIG IN (2, 21, 15, 75)   THEN 'DF'
    WHEN t.CODFILIALORIG IN (4, 41, 14, 74)   THEN 'UDI'
    WHEN t.CODFILIALORIG IN (7, 71)           THEN 'BH'
    WHEN t.CODFILIALORIG IN (10, 62, 13, 73)  THEN 'SP'
    WHEN t.CODFILIALORIG IN (3, 31, 16, 76)    THEN 'RJ'
    WHEN t.CODFILIALORIG IN (12, 72)          THEN 'CTBA'
    WHEN t.CODFILIALORIG IN (8, 81)           THEN 'POA'
		WHEN t.CODFILIALORIG IN (20)              THEN 'VALP'
    ELSE TO_CHAR(t.CODFILIALORIG)
  END AS ORIGEM,

  CASE
    WHEN t.CODFILIALDEST IN (1, 11, 5, 51)    THEN 'GO'
    WHEN t.CODFILIALDEST IN (2, 21, 15, 75)   THEN 'DF'
    WHEN t.CODFILIALDEST IN (4, 41, 14, 74)   THEN 'UDI'
    WHEN t.CODFILIALDEST IN (7, 71)           THEN 'BH'
    WHEN t.CODFILIALDEST IN (10, 62, 13, 73)  THEN 'SP'
    WHEN t.CODFILIALDEST IN (3, 31, 16, 76)   THEN 'RJ'
    WHEN t.CODFILIALDEST IN (12, 72)          THEN 'CTBA'
    WHEN t.CODFILIALDEST IN (8, 81)           THEN 'POA'
		WHEN t.CODFILIALDEST IN (20)              THEN 'VALP'
    ELSE TO_CHAR(t.CODFILIALDEST)
  END AS DESTINO,

  t.TRANSFESTOQUE               AS TRANSF_GERENCIAL,
  t.DATA                        AS DATA_SOLICITACAO,
  e.nome_guerra || ' (' || t.CODFUNC || ')'        AS SOLICITANTE,

  CASE t.POSICAO
    WHEN 'L' THEN '1/5 TRANSFERENCIA CRIADA'
    WHEN 'E' THEN '2/5 EM COLETA'
    WHEN 'C' THEN '3/5 EM CONFERENCIA'
    WHEN 'V' THEN '4/5 EM ROTA'
    WHEN 'T' THEN '5/5 CONCLUIDA'
    WHEN 'N' THEN 'CANCELADA'
    ELSE 'VERIFICAR'
  END AS STATUS_POSICAO,

  t.DATAPOSICAO                 AS DATA_ATUALIZACAO,
  d.nome_guerra || ' (' || t.CODFUNCPOSICAO || ')' AS FUNC_ATUALIZACAO

FROM   WINDOW.TAB_CONTROLE_TRANSF t
LEFT JOIN WINDOW.PCEMPR d ON d.matricula = t.codfuncposicao
LEFT JOIN WINDOW.PCEMPR e ON e.matricula = t.codfunc

WHERE  t.TRANSACAO >= 1442
ORDER BY t.DATA DESC
