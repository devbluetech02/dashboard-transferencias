SELECT
  i.TRANSACAO,
  i.CODPROD,
  p.DESCRICAO AS NOME_PRODUTO,
  i.QUANTIDADE,
  i.SUGESTAO,
  i.DATA,
  i.POSICAO,
  i.QTCOLETADA
FROM WINDOW.TAB_CONTROLE_ITEM_TRANSF i
LEFT JOIN WINDOW.PCPRODUT p ON i.CODPROD = p.CODPROD
WHERE
  {{ !List1.selectedItem ? '1 = 0' : '1 = 1' }}
  AND i.TRANSACAO >= 1442
  AND i.TRANSACAO = {{ Number(List1.selectedItem.TRANSACAO) }}
ORDER BY
  i.CODPROD
