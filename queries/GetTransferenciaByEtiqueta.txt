SELECT
  TRANSACAO,
	CODBARRAS,
  FILIALORIG,
  FILIALDEST,
  DATA,
  CODFUNC,
  CONFIRMADO,
  LEUCODBARRAS,
  CODFUNCLEITURA,
  DATALEITURA,
  PERMITEINCLUIR,
  DATACONFENT,
  CODFUNCCONFENT,
  DATACONFSAI,
  CODFUNCCONFSAI,
  TRANSFESTOQUE,
  VERSAOROTINA,
  CODPROD,
  QUANTIDADE
FROM WINDOW.TAB_TRANSF_ETIQUETA
WHERE TRANSACAO >= 1442
  AND (
    '{{input_etiqueta.text}}' IS NULL
    OR '{{input_etiqueta.text}}' = ''
    OR TO_CHAR(CODBARRAS) LIKE '%' || '{{input_etiqueta.text}}' || '%'
  )
ORDER BY CODBARRAS
